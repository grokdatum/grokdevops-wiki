$ ls -lZ /usr/bin/httpd
-rwxr-xr-x. 1 root root system_u:object_r:httpd_exec_t:s0 576400 Jan 15  2026 /usr/bin/httpd

# NOTE: Compare with httpd -- system binaries have proper exec types like httpd_exec_t.
# The inventory-api binary has usr_t which is not an executable type for system services.

$ sesearch --allow -s init_t -t usr_t -c file | grep execute
(no output)

# NOTE: No allow rule exists for init_t to execute files labeled usr_t.
# This confirms the denial is expected behavior under the targeted policy.

### Internal zone file: /etc/bind/zones/internal.acme.com.zone

@       IN      SOA     ns1.acme.com. admin.acme.com. (
                        2026021501      ; Serial
                        3600            ; Refresh
                        900             ; Retry
                        604800          ; Expire
                        300 )           ; Negative Cache TTL

        IN      NS      ns1.acme.com.
        IN      NS      ns2.acme.com.

ns1     IN      A       10.100.1.10
ns2     IN      A       10.100.1.11

; Application services

### named.conf forwarding configuration:
options {
    directory "/var/cache/bind";
    forwarders {
        8.8.8.8;
        8.8.4.4;
    };
    forward only;
    dnssec-validation auto;
};

# Because 'forward only' is set, queries for records NOT in the
# authoritative internal zone are forwarded to 8.8.8.8 -- this is
# why payments.acme.com resolves to the external IP.

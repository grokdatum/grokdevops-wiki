# Note: DHCPDISCOVER messages arrive via relay (via 10.20.0.1, via 10.30.0.1) for

### DHCP server pool config ###

$ cat /etc/dhcp/dhcpd.conf | grep -B1 -A5 "subnet 10.50"
subnet 10.50.0.0 netmask 255.255.255.0 {
  range 10.50.0.100 10.50.0.200;
  option routers 10.50.0.1;
  option domain-name-servers 10.10.10.2, 10.10.10.3;
  default-lease-time 86400;
}
# Pool exists -- the problem is purely that requests never reach the server.
